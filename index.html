const offlineKey='yard_offline_queue'; const offlineKey2='yard_sort_offline_queue';
function getQueue(k){ try{ return JSON.parse(localStorage.getItem(k)||'[]')}catch{return []}}
function setQueue(k,q){ localStorage.setItem(k, JSON.stringify(q)); updateOfflineUI();}
function updateOfflineUI(){
  const q1=getQueue(offlineKey); const q2=getQueue(offlineKey2); const total=q1.length+q2.length;
  const el=document.getElementById('offlineCount'); const sync=document.getElementById('syncBtn');
  if(total>0){ el.textContent=total+' queued'; el.classList.remove('hidden'); sync.classList.remove('hidden'); } else { el.classList.add('hidden'); sync.classList.add('hidden'); }
}

  document.getElementById('onlineStatus').textContent=online?'Online':'Offline';
  document.getElementById('onlineStatus').className='px-2 py-1 rounded text-xs '+(online?'bg-green-500':'bg-red-500');
  if(online) { syncQueue(); syncSortQueue(); }
}

async function fileToBase64(file){ return new Promise(res=>{ const r=new FileReader(); r.onload=()=>res(r.result); r.readAsDataURL(file); }); }
function base64ToFile(b64,name){ const arr=b64.split(','), mime=arr[0].match(/:(.*?);/)[1], bstr=atob(arr[1]); let n=bstr.length, u8=new Uint8Array(n); while(n--) u8[n]=bstr.charCodeAt(n); return new File([u8], name, {type:mime}); }
async function saveInbound(data){
  try{
    let photoFile=data._photoFile || (data._photoBase64?base64ToFile(data._photoBase64,data._photoName||'scale.jpg'):null);
    let photoUrl=null;
    if(photoFile){ const fileName=Date.now()+'_'+photoFile.name; const {error:upErr}=await supabase.storage.from('yard-photos').upload(fileName, photoFile); if(upErr) throw upErr; const {data:pub}=supabase.storage.from('yard-photos').getPublicUrl(fileName); photoUrl=pub.publicUrl; }
    const {data:inserted,error}=await supabase.from('yard_inbound').insert({ scale_id:data.scale_id, weight_kg:data.weight_kg, scale_qr:data.scale_qr, photo_url:photoUrl }).select().single();
    if(error) throw error; alert('Saved '+inserted.ticket_no); loadInbound(); loadInboundSelect();
  }catch(err){ console.error(err); alert('Save failed: '+err.message+' - queued'); const q=getQueue(offlineKey); q.push(data); setQueue(offlineKey,q); }
}
async function syncQueue(){
  const q=getQueue(offlineKey); if(q.length===0) return;
  document.getElementById('syncBtn').textContent='Syncing '+q.length+'...';
  const remaining=[];
  for(const item of q){ try{ await saveInbound(item); } catch{ remaining.push(item); } }
  setQueue(offlineKey,remaining); document.getElementById('syncBtn').textContent='Sync Now'; loadInbound(); loadInboundSelect();
}
async function loadInbound(){

}

// Sorting logic
let inboundMap={};
async function loadInboundSelect(){
  const {data}=await supabase.from('yard_inbound').select('id,ticket_no,weight_kg,scale_id').order('created_at',{ascending:false}).limit(100);
  const sel=document.getElementById('inbound_select');
  if(!data||data.length===0){ sel.innerHTML='<option value="">No YB-IN tickets yet - create one first</option>'; return; }
  // filter out already sorted? show all for now but mark sorted
  const {data:sorted}=await supabase.from('yard_sorting').select('inbound_id');
  const sortedIds=new Set((sorted||[]).map(s=>s.inbound_id));
  sel.innerHTML='<option value="">Select YB-IN ticket</option>'+data.map(r=>{
    const isSorted=sortedIds.has(r.id);
    inboundMap[r.id]=r;
    return `<option value="${r.id}" ${isSorted?'disabled':''}>${r.ticket_no} - ${r.weight_kg}kg ${r.scale_id} ${isSorted?'(already sorted)':''}</option>`;
  }).join('');
  // also store in map
  data.forEach(r=>{inboundMap[r.ticket_no]=r; inboundMap[r.id]=r;});
}
document.getElementById('inbound_select')?.addEventListener('change', e=>{
  const id=e.target.value; const r=inboundMap[id];
  const info=document.getElementById('inboundInfo');
  if(!r){ info.classList.add('hidden'); calc(); return; }
  info.classList.remove('hidden');
  info.innerHTML=`<b>${r.ticket_no}</b> - Inbound Weight: <b>${r.weight_kg} kg</b> from ${r.scale_id}`;
  calc();
});
function calc(){
  const sel=document.getElementById('inbound_select').value;
  const inbound = inboundMap[sel]?.weight_kg || 0;
  const waste=parseFloat(document.getElementById('waste_kg').value)||0;
  const clean=parseFloat(document.getElementById('clean_kg').value)||0;
  const sum=waste+clean;
  const loss=inbound-sum;
  const pct=inbound>0?(loss/inbound*100):0;
  document.getElementById('calcInbound').textContent=inbound?inbound.toFixed(2)+' kg':'-';
  document.getElementById('calcSum').textContent=sum.toFixed(2)+' kg';
  document.getElementById('calcLoss').textContent=loss.toFixed(2)+' kg';
  document.getElementById('calcPct').textContent=pct.toFixed(2)+' %';
  const status=document.getElementById('calcStatus');
  const btn=document.getElementById('saveSortBtn');
  if(inbound===0){ status.textContent='Select YB-IN first'; status.className='mt-2 text-xs font-bold p-2 rounded text-center bg-slate-100'; btn.disabled=true; btn.className='w-full bg-slate-300 text-white py-3 rounded font-bold'; btn.textContent='Select YB-IN first'; return; }
  if(sum===0){ status.textContent='Enter Waste + Clean'; status.className='mt-2 text-xs font-bold p-2 rounded text-center bg-slate-100'; btn.disabled=true; btn.className='w-full bg-slate-300 text-white py-3 rounded font-bold'; return; }
  if(loss<0){ status.textContent='❌ Waste + Clean > Inbound! Reduce.'; status.className='mt-2 text-xs font-bold p-2 rounded text-center bg-red-100 text-red-700'; btn.disabled=true; btn.className='w-full bg-red-300 text-white py-3 rounded font-bold'; btn.textContent='Invalid - Fix weights'; return; }
  if(pct>=2){ status.textContent=`❌ Loss ${pct.toFixed(2)}% >=2% - NOT ALLOWED`; status.className='mt-2 text-xs font-bold p-2 rounded text-center bg-red-100 text-red-700'; btn.disabled=true; btn.className='w-full bg-red-400 text-white py-3 rounded font-bold'; btn.textContent=`Loss too high (${pct.toFixed(2)}%)`; return; }
  status.textContent=`✅ OK - Loss ${pct.toFixed(2)}% <2% - Can save`; status.className='mt-2 text-xs font-bold p-2 rounded text-center bg-green-100 text-green-700'; btn.disabled=false; btn.className='w-full bg-green-600 text-white py-3 rounded font-bold'; btn.textContent=`SAVE YB-SORT - Loss ${pct.toFixed(2)}%`;
}
document.getElementById('waste_kg')?.addEventListener('input', calc);
document.getElementById('clean_kg')?.addEventListener('input', calc);

}

  for(const item of q){ try{ await saveSorting(item); } catch{ remaining.push(item); } }
  setQueue(offlineKey2,remaining);
}
async function loadSorting(){
  const {data}=await supabase.from('yard_sorting').select('*').order('created_at',{ascending:false}).limit(20);
  const list=document.getElementById('sortingList');
  if(!data||data.length===0){ list.innerHTML='<p class="text-xs text-slate-400">No sorting yet</p>'; return; }

}
